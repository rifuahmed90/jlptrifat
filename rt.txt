    '/css/styles.css',
    '/css/responsive.css',
    '/js/app.js',

    '/js/data.js',

  y=Outfit:wght@400;500;600;700;800&display=swap'
  ];

  self.addEventListener('install', event => {
    event.waitUntil(
      caches.open(CACHE_NAME)
        .then(cache => cache.addAll(urlsToCache))
    );
  });

  self.addEventListener('fetch', event => {
    event.respondWith(
      caches.match(event.request)
        .then(response => response || fetch(event.request))
    );
  });

  ধাপ১: ক্যাশকরুন সব ফাইল

  // js/offline-manager.js - আপনারযুক্ত করুন
  class OfflineManager {
    constructor() {
      this.cacheName = 'openjlpt-offline-v1';
      this.filesToCache = [
        '/index.html',
        '/css/styles.css',
        '/css/responsive.css',
        '/js/app.js',
        '/js/views.js',
        '/js/data.js',
        '/js/firebase-config.js',
        '/js/firebase-sync.js',
        '/manifest.json'
      ];
    }

    async init() {
      if ('serviceWorker' in navigator) {
        try {
          const cache = await caches.open(this.cacheName);
          await cache.addAll(this.filesToCache);
          console.log('✅ অফলাইনক্যাশসেভকরাহয়েছে');
        } catch (e) {
          console.error('❌ ক্যাশকরতেব্যর্থ:', e);
        }
      }
    }

    // অফলাইনডেটসেভকরা
    async saveOfflineData() {
      const backup = {
        learned: state.learned || [],
        favorites: state.favorites || [],
        quiz: state.quiz || { correct: 0, total: 0 },
        best: state.best || 0,
        days: state.days || [],
        prefs: state.prefs || {},
        srs: state.srs || {}
      };

      try {
        localStorage.setItem('openjlpt-offline-backup', JSON.stringify(backup));
        console.log('💾 অফলাইনব্যাকআপসেভকরাহয়েছে');
      } catch (e) {
        console.error('ব্যাকআপসেভকরতেব্যর্থ:', e);
      }
    }
  }

  // অটোমেটচালুকরুন
  document.addEventListener('DOMContentLoaded', () => {
    const offlineMgr = new OfflineManager();
    offlineMgr.init();

    // প্রতি৫ মিনিব্যাকআপসেভকরুন
    setInterval(() => offlineMgr.saveOfflineData(), 300000);
  });

  📦 ধাপ২: Bundle করারScript (build.js)

  build.js ফাইলতৈরকরুন:

  // build.js - সরল static bundler
  import fs from 'fs';
  import path from 'path';
  import { fileURLToPath } from 'url';
  import { readFileSync, writeFileSync } from 'fs';

  const __dirname = path.dirname(fileURLToPath(import.meta.url));

  function buildStatic() {
    console.log('🏗️  Building optimized static bundle...');

    // আপনারJS ফাইলগুলপঢ়ুন
    const appJs = readFileSync(path.join(__dirname, 'js/app.js'), 'utf8');
    const viewsJs = readFileSync(path.join(__dirname, 'js/views.js'), 'utf8');
    const dataJs = readFileSync(path.join(__dirname, 'js/data.js'), 'utf8');
    const firebaseSync = readFileSync(path.join(__dirname, 'js/firebase-sync.js'), 'utf8');

    // Bundle করুন সবগুছোটJS একসাথে
    const bundle = `
  /* Auto-generated bundle - OpenJLPT */
  ${dataJs}
  ${viewsJs}
  ${firebaseSync}
  ${appJs}
  `;

    // সেভকরুন
    writeFileSync(path.join(__dirname, 'dist/js/bundle.js'), bundle);

    // HTML আপডেটকরুন
    let html = readFileSync(path.join(__dirname, 'index.html'), 'utf8');
    html = html.replace('<script src="/js/app.js"></script>', '');
    html = html.replace('<script src="js/bundle.js"></script>', '');
    writeFileSync(path.join(__dirname, 'dist/index.html'), html);

    // CSS কপিকরুন
    fs.mkdirSync(path.join(__dirname, 'dist/css'), { recursive: true });
    writeFileSync(path.join(__dirname, 'dist/css/styles.css'), readFileSync(path.join(__dirname, 'css/styles.css'), 'utf8'));
    writeFileSync(path.join(__dirname, 'dist/css/responsive.css'), readFileSync(path.join(__dirname, 'css/responsive.css'),
  'utf8'));

    // আইকন কপি
    fs.cpSync(path.join(__dirname, 'icons'), path.join(__dirname, 'dist/icons'), { recursive: true });

    // সার্ভিওয়ার্কার
    writeFileSync(path.join(__dirname, 'dist/sw.js'), readFileSync(path.join(__dirname, 'sw.js'), 'utf8'));

    // ম্যানিফেস্ট
    const manifest = {
      name: "OpenJLPT Bangla",
      short_name: "JLPT",
      start_url: "/",
      display: "standalone",
      background_color: "#0a0e12",
      theme_color: "#0a0e12",
      icons: [
        { src: "/icons/icon-192.png", sizes: "192x192", type: "image/png" },
        { src: "/icons/icon-512.png", sizes: "512x512", type: "image/png" }
      ]
    };
    writeFileSync(path.join(__dirname, 'dist/manifest.json'), JSON.stringify(manifest, null, 2));

    console.log('✅ Bundle completed!');
    console.log('📁 Output: /dist folder ready for deployment');
  }

  // চালানnode build.js
  if (import.meta.url === `file://${process.argv[1]}`) {
    buildStatic();
  }

  export default buildStatic;

  // deploy.js - GitHub Pages এ ডিপ্লয়
  import { execSync } from 'child_process';
  import fs from 'fs';
  import path from 'path';
  import { fileURLToPath } from 'url';

  const __dirname = path.dirname(fileURLToPath(import.meta.url));

  function deploy() {
    try {
      // Build করুন
      execSync('node build.js', { stdio: 'inherit' });

      // ডিস্টফোল্ডাকপিকরুন
      execSync('cp -r dist/* .', { stdio: 'inherit' });

      // Git commit এবংpush
      execSync('git add .', { stdio: 'inherit' });
      execSync('git commit -m "Deploy: Auto build ' + new Date().toISOString() + '"', { stdio: 'inherit' });
      execSync('git push origin main', { stdio: 'inherit' });

      console.log('🎉 Deployed successfully!');
    } catch (error) {
      console.error('Deployment failed:', error);
    }
  }

  // চালানnode deploy.js
  if (import.meta.url === `file://${process.argv[1]}`) {
    deploy();
  }

  export default deploy;

  ২. সর্বশেষsw.js (আপনারমৌজুদারিপ্লেকরুন)

  const CACHE_NAME = 'jlptrifat-v1';
  const ASSETS = [
    '/',
    '/index.html',
    '/css/styles.css',
    '/css/responsive.css',
    '/js/app.js',
    '/js/data.js',
    '/js/firebase-config.js',
    '/js/firebase-sync.js',
    '/js/views.js',
    '/manifest.json',
    '/icons/icon.svg',
    '/icons/icon-180.png',
    '/icons/icon-192.png',
    '/icons/icon-512.png',
    '/icons/icon-maskable.svg',
    '/icons/icon-maskable-512.png'
  ];

  self.addEventListener('install', (event) => {
    event.waitUntil(
      caches.open(CACHE_NAME)
        .then((cache) => cache.addAll(ASSETS))
        .then(() => self.skipWaiting())
    );
  });

  self.addEventListener('activate', (event) => {
    event.waitUntil(
      caches.keys().then((cacheNames) => {
        return Promise.all(
          cacheNames.map((cacheName) => {
            if (cacheName !== CACHE_NAME) {
              return caches.delete(cacheName);
            }
          })
        );
      }).then(() => self.clients.claim())
    );
  });

  self.addEventListener('fetch', (event) => {
    if (event.request.method !== 'GET') return;

    event.respondWith(
      caches.match(event.request)
        .then((response) => response || fetch(event.request).then((networkResponse) => {
          if (!networkResponse || networkResponse.status !== 0 || networkResponse.type !== 'basic') {
            return networkResponse;
          }
          const responseToCache = networkResponse.clone();
          caches.open(CACHE_NAME)
            .then((cache) => cache.put(event.request, responseToCache));
          return networkResponse;
        }))
        .catch(() => fetch(event.request))
    );
  });

  self.addEventListener('sync', (event) => {
    if (event.tag === 'save-offline-data') {
      event.waitUntil(saveOfflineData());
    }
  });

  async function saveOfflineData() {
    try {
      const cachedState = await caches.match('/state.json');
      if (cachedState) {
        const state = await cachedState.json();
        localStorage.setItem('kanji-dark-v1', JSON.stringify(state));
      }
    } catch (e) {
      console.log('Background sync error', e);
    }
  }
